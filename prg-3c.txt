// C Program to simulate Proportional Scheduling
// (Lottery Scheduling Technique)

#include <stdio.h>
#include <stdlib.h>
#include <time.h>

struct Process
{
    int pid;
    int tickets;
    int burstTime;
};

int main()
{
    int n, i, totalTickets = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    struct Process p[n];

    // Input process details
    for(i = 0; i < n; i++)
    {
        p[i].pid = i + 1;

        printf("\nEnter tickets for Process P%d: ", p[i].pid);
        scanf("%d", &p[i].tickets);

        printf("Enter burst time for Process P%d: ", p[i].pid);
        scanf("%d", &p[i].burstTime);

        totalTickets += p[i].tickets;
    }

    srand(time(0));

    printf("\n--- Proportional (Lottery) Scheduling ---\n");

    int completed = 0;

    while(completed < n)
    {
        int winner = rand() % totalTickets + 1;

        int sum = 0;

        for(i = 0; i < n; i++)
        {
            sum += p[i].tickets;

            if(winner <= sum)
            {
                if(p[i].burstTime > 0)
                {
                    printf("Process P%d is executing\n", p[i].pid);

                    p[i].burstTime--;

                    if(p[i].burstTime == 0)
                    {
                        printf("Process P%d completed\n", p[i].pid);
                        completed++;
                    }
                }
                break;
            }
        }
    }

    printf("\nAll processes completed successfully.\n");

    return 0;
}
